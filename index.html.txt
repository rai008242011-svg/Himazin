import React, { useEffect, useRef, useState } from "react";
import { motion } from "framer-motion";

export default function ClockApp() {
  const [time, setTime] = useState(new Date());

  const [glowColor, setGlowColor] = useState("#a855f7");
  const [glowColor2, setGlowColor2] = useState("#3b82f6");
  const [bgColor, setBgColor] = useState("#0a0a0a");
  const [bgColor2, setBgColor2] = useState("#020617");
  const [textColor, setTextColor] = useState("#000000");

  const [savedThemes, setSavedThemes] = useState([]);

  const canvasRef = useRef(null);

  // load saved themes
  useEffect(() => {
    const data = localStorage.getItem("themes");
    if (data) setSavedThemes(JSON.parse(data));
  }, []);

  // clock
  useEffect(() => {
    const timer = setInterval(() => setTime(new Date()), 1000);
    return () => clearInterval(timer);
  }, []);

  // particles
  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    let particles = [];

    const resize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    };
    resize();
    window.addEventListener("resize", resize);

    const spawn = (x, y, power = 1) => {
      for (let i = 0; i < 20 * power; i++) {
        particles.push({
          x,
          y,
          dx: (Math.random() - 0.5) * 4,
          dy: (Math.random() - 0.5) * 4,
          life: 80 + Math.random() * 40,
        });
      }
    };

    window.addEventListener("mousemove", (e) => {
      spawn(e.clientX, e.clientY, 0.3);
    });

    window.addEventListener("click", (e) => {
      spawn(e.clientX, e.clientY, 1.5);
    });

    const draw = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      particles.forEach((p, i) => {
        p.x += p.dx;
        p.y += p.dy;
        p.life--;

        ctx.beginPath();
        ctx.arc(p.x, p.y, 2 + Math.random() * 2, 0, Math.PI * 2);
        ctx.fillStyle = glowColor;
        ctx.fill();

        if (p.life <= 0) particles.splice(i, 1);
      });

      requestAnimationFrame(draw);
    };

    draw();

    return () => {
      window.removeEventListener("resize", resize);
    };
  }, [glowColor]);

  const saveTheme = () => {
    const newTheme = { glowColor, glowColor2, bgColor, bgColor2 };
    const updated = [...savedThemes, newTheme];
    setSavedThemes(updated);
    localStorage.setItem("themes", JSON.stringify(updated));
  };

  const applyTheme = (theme) => {
    setGlowColor(theme.glowColor);
    setGlowColor2(theme.glowColor2);
    setBgColor(theme.bgColor);
    setBgColor2(theme.bgColor2);
  };

  const deleteTheme = (index) => {
    const updated = savedThemes.filter((_, i) => i !== index);
    setSavedThemes(updated);
    localStorage.setItem("themes", JSON.stringify(updated));
  };

  const hours = time.getHours().toString().padStart(2, "0");
  const minutes = time.getMinutes().toString().padStart(2, "0");
  const seconds = time.getSeconds().toString().padStart(2, "0");

  const date = time.toLocaleDateString(undefined, {
    year: "numeric",
    month: "long",
    day: "numeric",
    weekday: "long",
  });

  return (
    <div
      className="h-screen w-full flex flex-col items-center justify-center text-black overflow-hidden"
      style={{
        background: `linear-gradient(135deg, ${bgColor}, ${bgColor2})`,
      }}
    >
      <canvas ref={canvasRef} className="absolute inset-0 z-0" />

      {/* color controls */}
      <div className="absolute top-4 z-10 flex gap-2 items-center">
        <input type="color" value={glowColor} onChange={(e)=>setGlowColor(e.target.value)} />
        <input type="color" value={glowColor2} onChange={(e)=>setGlowColor2(e.target.value)} />
        <input type="color" value={bgColor} onChange={(e)=>setBgColor(e.target.value)} />
        <input type="color" value={bgColor2} onChange={(e)=>setBgColor2(e.target.value)} />
        <input type="color" value={textColor} onChange={(e)=>setTextColor(e.target.value)} title="文字色" />

        <button
          onClick={saveTheme}
          className="ml-2 px-3 py-1 bg-white/20 rounded-lg hover:bg-white/30"
        >
          保存
        </button>
      </div>

      {/* saved themes */}
      <div className="absolute bottom-4 z-10 flex gap-2 flex-wrap max-w-[90%] justify-center">
        {savedThemes.map((theme, i) => (
          <div key={i} className="flex items-center gap-1">
            <button
              onClick={() => applyTheme(theme)}
              className="w-8 h-8 rounded-lg border border-white/30"
              style={{
                background: `linear-gradient(135deg, ${theme.bgColor}, ${theme.bgColor2})`,
              }}
            />
            <button
              onClick={() => deleteTheme(i)}
              className="text-xs text-red-400"
            >
              ×
            </button>
          </div>
        ))}
      </div>

      <motion.div
        initial={{ scale: 0.8, opacity: 0 }}
        animate={{ scale: 1, opacity: 1 }}
        className="relative backdrop-blur-xl bg-white/10 border border-white/20 rounded-3xl p-10 shadow-2xl z-10 text-center"
        style={{ boxShadow: `0 0 40px ${glowColor}` }}
      >
        <div className="text-6xl font-mono tracking-widest" style={{ color: textColor }}>
          {hours}:{minutes}:{seconds}
        </div>

        <div className="mt-4 text-gray-300 text-lg">
          {date}
        </div>
      </motion.div>
    </div>
  );
}
